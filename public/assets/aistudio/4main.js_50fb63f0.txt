// ── collision ────────────────────────────────────────────────────────────────
function circleBox(p, r, b) {
  const cx = clamp(p.x, b.min.x, b.max.x), cz = clamp(p.z, b.min.z, b.max.z);
  const dx = p.x - cx, dz = p.z - cz, d2 = dx * dx + dz * dz;
  if (d2 >= r * r) return;
  if (d2 > 1e-8) { const d = Math.sqrt(d2), k = (r - d) / d; p.x += dx * k; p.z += dz * k; }
  else {
    const e = [p.x - b.min.x + r, b.max.x - p.x + r, p.z - b.min.z + r, b.max.z - p.z + r];
    const m = Math.min(...e);
    if (m === e[0]) p.x = b.min.x - r; else if (m === e[1]) p.x = b.max.x + r;
    else if (m === e[2]) p.z = b.min.z - r; else p.z = b.max.z + r;
  }
}
function resolve(p) {
  const r = GAME_CONFIG.player.radius, B = room.bounds;
  p.x = clamp(p.x, B.minX + r, B.maxX - r);
  p.z = clamp(p.z, B.minZ + r, B.maxZ - r);
  for (const b of room.colliders) circleBox(p, r, b);
}

const CC = GAME_CONFIG.camera;

function clampCam(v) {
  const m = CC.wallMargin;
  v.x = clamp(v.x, room.bounds.minX + m, room.bounds.maxX - m);
  v.z = clamp(v.z, room.bounds.minZ + m, room.bounds.maxZ - m);
  v.y = clamp(v.y, .5, GAME_CONFIG.room.height - .25);
}
function cameraTarget() {
  if (gameState === 'inspect') {
    camTarget.set(position.x + Math.sin(inspectAngle) * inspectR, inspectY, position.z + Math.cos(inspectAngle) * inspectR);
  } else {
    const fx = Math.sin(rotationY), fz = Math.cos(rotationY);
    camTarget.set(position.x - fx * CC.distance, CC.height, position.z - fz * CC.distance);
  }
  clampCam(camTarget);
}
cameraTarget(); camera.position.copy(camTarget);
function dampAngle(a, b, k) {
  let d = (b - a) % (Math.PI * 2);
  if (d > Math.PI) d -= Math.PI * 2; if (d < -Math.PI) d += Math.PI * 2;
  return a + d * k;
}

function drawRadar() {
  const c = radar, W = 150, H = 150, B = room.bounds;
  const sx = (x) => (x - B.minX) / (B.maxX - B.minX) * (W - 12) + 6;
  const sz = (z) => (z - B.minZ) / (B.maxZ - B.minZ) * (H - 12) + 6;
  c.clearRect(0, 0, W, H);
  c.fillStyle = '#0b1220'; c.fillRect(0, 0, W, H);
  c.strokeStyle = '#2a3c56'; c.lineWidth = 1; c.strokeRect(6, 6, W - 12, H - 12);
  c.fillStyle = '#182437';
  for (const b of room.colliders) c.fillRect(sx(b.min.x), sz(b.min.z), sx(b.max.x) - sx(b.min.x), sz(b.max.z) - sz(b.min.z));
  const cm = mission.cam;
  const ang = Math.PI / 2 - cm.yaw;
  const rPx = cm.reach / (B.maxX - B.minX) * (W - 12);
  c.beginPath(); c.moveTo(sx(cm.x), sz(cm.z));
  c.arc(sx(cm.x), sz(cm.z), rPx, ang - cm.half, ang + cm.half);
  c.closePath();
  c.fillStyle = mission.spotted ? 'rgba(255,59,92,.32)' : 'rgba(255,180,84,.18)'; c.fill();
  c.beginPath(); c.arc(sx(cm.x), sz(cm.z), 3, 0, 7); c.fillStyle = '#ffb454'; c.fill();
  for (const it of mission.intel) {
    if (!it.active) continue;
    c.beginPath(); c.arc(sx(it.x), sz(it.z), 3.2, 0, 7); c.fillStyle = '#38c6ff'; c.fill();
  }
  const E = mission.exitPos;
  c.beginPath(); c.arc(sx(E.x), sz(E.z), 5, 0, 7);
  c.strokeStyle = mission.exitActive ? '#a4f45c' : '#5f7089'; c.lineWidth = 2; c.stroke();
  c.save(); c.translate(sx(position.x), sz(position.z)); c.rotate(Math.PI - rotationY);
  c.beginPath(); c.moveTo(0, -6); c.lineTo(4.5, 5); c.lineTo(-4.5, 5); c.closePath();
  c.fillStyle = '#ece7db'; c.fill(); c.restore();
}

function start(lv) {
  level = Math.max(1, lv);
  mission.reset(level);
  position.set(spawn.x, spawn.y, spawn.z); rotationY = spawn.rotationY;
  agent.setTransform(position, rotationY); agent.setAnimationState('idle');

const clock = new THREE.Clock();
let netAcc = 0, hudAcc = 0;
const netInterval = 1 / GAME_CONFIG.network.tickRate;

function frame() {
  requestAnimationFrame(frame);
  const dt = Math.min(clock.getDelta(), .05);
  const t = clock.elapsedTime;
  const P = GAME_CONFIG.player;
  const playing = gameState === 'playing';

  const k = input.getIntent();
  const intent = playing ? {
    forward: clamp(k.forward + touch.f - touch.b, -1, 1),
    turn: clamp(k.turn + touch.l - touch.r, -1, 1),
    sprint: k.sprint || touch.s === 1,
  } : { forward: 0, turn: 0, sprint: false };

  if (playing) {
    rotationY += intent.turn * P.turnSpeed * dt;
    let sp = intent.sprint ? P.runSpeed : P.walkSpeed;
    if (intent.forward < 0) sp *= P.backwardFactor;
    position.x += Math.sin(rotationY) * intent.forward * sp * dt;
    position.z += Math.cos(rotationY) * intent.forward * sp * dt;
    resolve(position);
  }

  const moving = intent.forward !== 0;
  const state = moving ? (intent.sprint ? 'run' : 'walk') : 'idle';
  agent.setTransform(position, rotationY);
  agent.setAnimationState(state, intent.forward < 0 ? -1 : 1);
  agent.update(dt);
  if (agent.consumeFootstep()) audio.footstep(intent.sprint);

  mission.update(dt, position, moving, intent.sprint, playing);

  if (gameState === 'inspect' && !dragging) inspectAngle += dt * .18;
  cameraTarget();
  camera.position.lerp(camTarget, 1 - Math.exp(-(gameState === 'inspect' ? 9 : CC.followSmoothing) * dt));
  if (shake > .001) {
    shake = Math.max(0, shake - dt * 1.6);
    camera.position.x += (Math.random() - .5) * shake * .5;
    camera.position.y += (Math.random() - .5) * shake * .4;
  }
  lookAt.set(position.x, gameState === 'inspect' ? 1.32 : CC.lookAtHeight, position.z);
  camera.lookAt(lookAt);

  const dp = dustGeo.attributes.position;
  for (let i = 0; i < DUST; i++) {
    let y = dp.getY(i) + dustSpd[i] * dt;
    if (y > 3.7) y = .05;
    dp.setY(i, y);
    dp.setX(i, dp.getX(i) + Math.sin(t * .35 + i) * .0007);
  }
  dp.needsUpdate = true;
  rays.rotation.y = Math.sin(t * .1) * .04;
  rayMat.opacity = .045 + Math.sin(t * .7) * .012;
  fillLight.intensity = 8 + Math.sin(t * .9) * 1.4;

  const rk = 1 - Math.exp(-12 * dt);
  for (const r of remotes.values()) {
    r.cur.lerp(r.tgt, rk); r.rot = dampAngle(r.rot, r.trot, rk);
    r.a.setTransform(r.cur, r.rot); r.a.update(dt);
  }

  netAcc += dt;
  if (netAcc >= netInterval) {
    netAcc = 0;
    network.broadcastState({ position: { x: position.x, y: position.y, z: position.z }, rotationY, animationState: state, sentAt: Date.now() });
  }
  hudAcc += dt;
  if (hudAcc >= .08) { hudAcc = 0; updateHUD(); drawRadar(); }

  renderer.render(scene, camera);
}

pips(mission.total);
