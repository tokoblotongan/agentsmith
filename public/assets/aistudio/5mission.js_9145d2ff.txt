import * as THREE from 'three';
import { GAME_CONFIG } from '../config/gameConfig.js';

const MC = GAME_CONFIG.mission;

function glowTexture() {
  const s = 128, c = document.createElement('canvas'); c.width = c.height = s;
  const x = c.getContext('2d'), g = x.createRadialGradient(s / 2, s / 2, 0, s / 2, s / 2, s / 2);
  g.addColorStop(0, 'rgba(255,255,255,1)'); g.addColorStop(.35, 'rgba(255,255,255,.45)'); g.addColorStop(1, 'rgba(255,255,255,0)');
  x.fillStyle = g; x.fillRect(0, 0, s, s);
  const t = new THREE.CanvasTexture(c); t.colorSpace = THREE.SRGBColorSpace; return t;
}
function segBlocked(ax, az, bx, bz, b) {
  const dx = bx - ax, dz = bz - az;
  let t0 = 0, t1 = 1;
  const axes = [[ax, dx, b.min.x, b.max.x], [az, dz, b.min.z, b.max.z]];
  for (const [p, d, mn, mx] of axes) {
    if (Math.abs(d) < 1e-8) { if (p < mn || p > mx) return false; }
    else {
      let ta = (mn - p) / d, tb = (mx - p) / d;
      if (ta > tb) { const s = ta; ta = tb; tb = s; }
      t0 = Math.max(t0, ta); t1 = Math.min(t1, tb);
      if (t0 > t1) return false;
    }
  }
  return true;
}

export class Mission {
  constructor(scene, room, hooks = {}) {
    this.hooks = hooks;
    this.blockers = room.colliders.filter((b) => b.max.y >= 1.1);
    this.glow = glowTexture();
    this.level = 1; this.state = 'idle';
    this.collected = 0; this.total = 4; this.detection = 0; this.timeLeft = 120;
    this.exitActive = false; this.score = 0; this.alerted = false; this.spotted = false;
    this._t = 0; this._alertClock = 0;
    this._buildIntel(scene);
    this._buildCamera(scene);
    this._buildExit(scene);
    this._buildParticles(scene);
    this.reset(1);
    this.state = 'idle';
  }

  _buildIntel(scene) {
    this.intel = MC.intelSpots.map(([x, z]) => {
      const g = new THREE.Group(); g.position.set(x, 0.95, z);
      const body = new THREE.Mesh(new THREE.BoxGeometry(.30, .035, .22),
        new THREE.MeshStandardMaterial({ color: 0x14202f, roughness: .5, metalness: .3 }));
      body.castShadow = true; g.add(body);
      const label = new THREE.Mesh(new THREE.BoxGeometry(.20, .006, .12),
        new THREE.MeshStandardMaterial({ color: 0x0a1a26, emissive: 0x38c6ff, emissiveIntensity: 2.2, roughness: .3 }));
      label.position.set(0, .022, 0); g.add(label);
      const ring = new THREE.Mesh(new THREE.TorusGeometry(.26, .008, 8, 28),
        new THREE.MeshBasicMaterial({ color: 0x38c6ff, transparent: true, opacity: .55 }));
      ring.rotation.x = Math.PI / 2; ring.position.y = -.10; g.add(ring);
      const sp = new THREE.Sprite(new THREE.SpriteMaterial({ map: this.glow, color: 0x38c6ff, transparent: true, opacity: .55, blending: THREE.AdditiveBlending, depthWrite: false }));
      sp.scale.set(1.1, 1.1, 1); g.add(sp);
      scene.add(g);
      return { group: g, ring, label, x, z, active: true, phase: Math.random() * 6.28 };
    });
  }

  _buildCamera(scene) {
    const C = MC.camera;
    this.cam = { x: C.x, y: C.y, z: C.z, yaw: 0, half: C.halfAngle, reach: C.reach, speed: .34 };
    const rig = new THREE.Group(); rig.position.set(C.x, C.y, C.z); scene.add(rig);
    const mount = new THREE.Mesh(new THREE.BoxGeometry(.16, .16, .12),
      new THREE.MeshStandardMaterial({ color: 0x1a2230, roughness: .6, metalness: .5 }));
    rig.add(mount);
    this.head = new THREE.Group(); rig.add(this.head);
    const shell = new THREE.Mesh(new THREE.BoxGeometry(.20, .16, .30),
      new THREE.MeshStandardMaterial({ color: 0x222c3c, roughness: .45, metalness: .6 }));
    shell.position.z = .16; shell.castShadow = true; this.head.add(shell);
    this.lens = new THREE.Mesh(new THREE.CylinderGeometry(.055, .065, .06, 16),
      new THREE.MeshStandardMaterial({ color: 0x0a0f16, emissive: 0xffb454, emissiveIntensity: 2.4, roughness: .2 }));
    this.lens.rotation.x = Math.PI / 2; this.lens.position.z = .33; this.head.add(this.lens);

    const h = C.reach, r = Math.tan(C.halfAngle) * h;
    const cg = new THREE.ConeGeometry(r, h, 26, 1, true);
    cg.translate(0, -h / 2, 0); cg.rotateX(-Math.PI / 2);
    this.beamMat = new THREE.MeshBasicMaterial({ color: 0xffb454, transparent: true, opacity: .085, blending: THREE.AdditiveBlending, depthWrite: false, side: THREE.DoubleSide, fog: false });
    this.beam = new THREE.Mesh(cg, this.beamMat); this.head.add(this.beam);

    this.spot = new THREE.SpotLight(0xffb454, 26, C.reach + 2, C.halfAngle, .45, 2);
    this.spot.position.set(0, 0, .3);
    this.spotTarget = new THREE.Object3D(); this.spotTarget.position.set(0, -1.6, C.reach);
    this.head.add(this.spot, this.spotTarget); this.spot.target = this.spotTarget;
  }

  _buildExit(scene) {
    const E = MC.exit;
    this.exitPos = new THREE.Vector3(E.x, 0, E.z);
    const g = new THREE.Group(); g.position.set(E.x, 0, E.z); scene.add(g);
    this.exitRing = new THREE.Mesh(new THREE.RingGeometry(E.radius - .12, E.radius, 40),
      new THREE.MeshBasicMaterial({ color: 0x5f7089, transparent: true, opacity: .5, side: THREE.DoubleSide }));
    this.exitRing.rotation.x = -Math.PI / 2; this.exitRing.position.y = .02; g.add(this.exitRing);
    this.exitColumn = new THREE.Mesh(new THREE.CylinderGeometry(E.radius - .18, E.radius - .18, 2.4, 24, 1, true),
      new THREE.MeshBasicMaterial({ color: 0x5f7089, transparent: true, opacity: .07, blending: THREE.AdditiveBlending, depthWrite: false, side: THREE.DoubleSide }));
    this.exitColumn.position.y = 1.2; g.add(this.exitColumn);
    this.exitGlow = new THREE.Sprite(new THREE.SpriteMaterial({ map: this.glow, color: 0x5f7089, transparent: true, opacity: .35, blending: THREE.AdditiveBlending, depthWrite: false }));
    this.exitGlow.scale.set(2.2, 2.2, 1); this.exitGlow.position.y = .3; g.add(this.exitGlow);
    this.exitGroup = g;
  }

  _buildParticles(scene) {
    const N = this.N = 150;
    this.pPos = new Float32Array(N * 3); this.pVel = new Float32Array(N * 3); this.pLife = new Float32Array(N);
    for (let i = 0; i < N; i++) this.pPos[i * 3 + 1] = -999;
    const g = new THREE.BufferGeometry();
    g.setAttribute('position', new THREE.BufferAttribute(this.pPos, 3));
    this.points = new THREE.Points(g, new THREE.PointsMaterial({
      color: 0x9be9ff, size: .075, transparent: true, opacity: .95,
      blending: THREE.AdditiveBlending, depthWrite: false, map: this.glow,
    }));
    this.points.frustumCulled = false; scene.add(this.points);
    this.pCursor = 0;
  }

  burst(x, y, z, count = 26) {
    for (let k = 0; k < count; k++) {
      const i = this.pCursor = (this.pCursor + 1) % this.N;
      const a = Math.random() * Math.PI * 2, e = Math.random() * Math.PI * .5, sp = 1.2 + Math.random() * 2.2;
      this.pPos[i * 3] = x; this.pPos[i * 3 + 1] = y; this.pPos[i * 3 + 2] = z;
      this.pVel[i * 3] = Math.cos(a) * Math.cos(e) * sp;
      this.pVel[i * 3 + 1] = Math.sin(e) * sp + .8;
      this.pVel[i * 3 + 2] = Math.sin(a) * Math.cos(e) * sp;
      this.pLife[i] = .7 + Math.random() * .5;
    }
  }

  params(level) {
    const L = MC.levels[Math.min(level, 3) - 1];
    const extra = Math.max(0, level - 3);
    return {
      name: L.name + (extra ? ' +' + extra : ''),
      intel: Math.min(MC.intelSpots.length, L.intel + extra),
      time: Math.max(70, L.time - extra * 4),
      sweep: L.sweep * Math.pow(1.08, extra),
      detect: L.detect + extra * 3,
    };
  }

  reset(level) {
    this.level = level;
    const P = this.params(level);
    this.params_ = P;
    this.total = P.intel; this.collected = 0; this.detection = 0;
    this.timeLeft = P.time; this.exitActive = false; this.score = 0;
    this.alerted = false; this.spotted = false; this._alertClock = 0;
    this.cam.speed = P.sweep;
    const order = this.intel.map((_, i) => i);
    for (let i = order.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); const t = order[i]; order[i] = order[j]; order[j] = t; }
    this.intel.forEach((it, i) => {
      it.active = order.indexOf(i) < this.total;
      it.group.visible = it.active;
    });
    this._setExit(false);
    for (let i = 0; i < this.N; i++) { this.pLife[i] = 0; this.pPos[i * 3 + 1] = -999; }
    this.points.geometry.attributes.position.needsUpdate = true;
    this.state = 'playing';
  }

  _setExit(on) {
    this.exitActive = on;
    const c = on ? 0xa4f45c : 0x5f7089;
    this.exitRing.material.color.setHex(c);
    this.exitColumn.material.color.setHex(c);
    this.exitGlow.material.color.setHex(c);
    this.exitRing.material.opacity = on ? .85 : .4;
    this.exitColumn.material.opacity = on ? .16 : .05;
    this.exitGlow.material.opacity = on ? .6 : .25;
  }

  update(dt, player, moving, sprinting, active) {
    this._t += dt;

    // visuals always run so the scene stays alive on overlays
    for (const it of this.intel) {
      if (!it.active) continue;
      it.group.position.y = .95 + Math.sin(this._t * 1.8 + it.phase) * .07;
      it.group.rotation.y += dt * .9;
      it.ring.rotation.z += dt * 1.4;
      it.label.material.emissiveIntensity = 1.8 + Math.sin(this._t * 3 + it.phase) * .6;
    }
    this.cam.yaw = Math.sin(this._t * this.cam.speed) * MC.camera.range;
    this.head.rotation.y = this.cam.yaw;
    this.exitRing.rotation.z += dt * (this.exitActive ? 1.1 : .25);
    this.exitGlow.material.opacity = (this.exitActive ? .55 : .22) + Math.sin(this._t * 2.4) * .12;

    for (let i = 0; i < this.N; i++) {
      if (this.pLife[i] <= 0) continue;
      this.pLife[i] -= dt;
      if (this.pLife[i] <= 0) { this.pPos[i * 3 + 1] = -999; continue; }
      this.pVel[i * 3 + 1] -= 5.2 * dt;
      this.pPos[i * 3] += this.pVel[i * 3] * dt;
      this.pPos[i * 3 + 1] += this.pVel[i * 3 + 1] * dt;
      this.pPos[i * 3 + 2] += this.pVel[i * 3 + 2] * dt;
    }
    this.points.geometry.attributes.position.needsUpdate = true;

    if (!active || this.state !== 'playing') { this._paintBeam(0); return; }

    this.timeLeft -= dt;

    // pickups
    for (const it of this.intel) {
      if (!it.active) continue;
      const dx = player.x - it.x, dz = player.z - it.z;
      if (dx * dx + dz * dz < MC.pickupRadius * MC.pickupRadius) {
        it.active = false; it.group.visible = false; this.collected++;
        this.burst(it.x, 1.0, it.z, 30);
        if (this.hooks.onIntel) this.hooks.onIntel(this.collected, this.total);
        if (this.collected >= this.total) { this._setExit(true); if (this.hooks.onExit) this.hooks.onExit(); }
      }
    }

    // detection cone + line of sight
    const fx = Math.sin(this.cam.yaw), fz = Math.cos(this.cam.yaw);
    const dx = player.x - this.cam.x, dz = player.z - this.cam.z;
    const dist = Math.hypot(dx, dz);
    let inside = false;
    if (dist < this.cam.reach && dist > .001) {
      const dot = (dx / dist) * fx + (dz / dist) * fz;
      if (Math.acos(Math.max(-1, Math.min(1, dot))) < this.cam.half) {
        inside = !this.blockers.some((b) => segBlocked(this.cam.x, this.cam.z, player.x, player.z, b));
      }
    }
    this.spotted = inside;
    const rate = this.params_.detect * (sprinting && moving ? MC.sprintNoise : moving ? 1 : MC.stillFactor);
    this.detection = Math.max(0, Math.min(100, this.detection + (inside ? rate : -MC.decay) * dt));
    this._paintBeam(this.detection / 100);

    if (inside) {
      this._alertClock -= dt;
      if (this._alertClock <= 0) {
        this._alertClock = this.detection > 70 ? .28 : .6;
        if (this.hooks.onAlert) this.hooks.onAlert(this.detection / 100);
      }
    }
    const nowAlert = this.detection > 35;
    if (nowAlert !== this.alerted) { this.alerted = nowAlert; if (this.hooks.onAlertState) this.hooks.onAlertState(nowAlert); }

    if (this.detection >= 100) return this._lose('BUSTED');
    if (this.timeLeft <= 0) { this.timeLeft = 0; return this._lose('TIME'); }

    if (this.exitActive) {
      const ex = player.x - this.exitPos.x, ez = player.z - this.exitPos.z;
      if (ex * ex + ez * ez < MC.exit.radius * MC.exit.radius) this._win();
    }
  }

  _paintBeam(t) {
    const c = new THREE.Color().setHSL(THREE.MathUtils.lerp(.11, .98, Math.min(1, t * 1.2)), .95, .58);
    this.beamMat.color.copy(c);
    this.beamMat.opacity = .07 + t * .16 + (this.spotted ? .05 : 0);
    this.lens.material.emissive.copy(c);
    this.spot.color.copy(c);
    this.spot.intensity = 22 + t * 26;
  }

  _win() {
    this.state = 'won';
    const timeBonus = Math.floor(this.timeLeft) * 12;
    const stealth = Math.round((100 - this.detection) * 8);
    this.score = this.total * 1000 + timeBonus + stealth + (this.level - 1) * 500;
    const rank = this.score > this.total * 1000 + 1400 ? 'S' : this.score > this.total * 1000 + 900 ? 'A' : this.score > this.total * 1000 + 400 ? 'B' : 'C';
    this.burst(this.exitPos.x, 1.2, this.exitPos.z, 60);
    if (this.hooks.onWin) this.hooks.onWin({ score: this.score, rank, timeLeft: Math.floor(this.timeLeft), detection: Math.round(this.detection), level: this.level, intel: this.total });
  }

  _lose(reason) {
    this.state = 'lost';
    this.score = this.collected * 400;
    if (this.hooks.onLose) this.hooks.onLose({ reason, collected: this.collected, total: this.total, score: this.score, level: this.level });
  }
}
